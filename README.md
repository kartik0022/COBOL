ENVIRONMENT DIVISION.
INPUT-OUTPUT SECTION.
FILE-CONTROL.

DATA DIVISION.
FILE SECTION.
WORKING-STORAGE SECTION.
01 CARD-DETAILS-IN.
	03 	CARD-NUM 		PIC X(5).
	03	CARD-NAME		PIC X(10).
	03 	CARD-LIMIT			PIC S9(5).
	03 	CARD-DUE			PIC S9(05)V9(02).

01 CARD-DETAILS-OUT.
	03 	CARD-NUM 		PIC X(5)        VALUE  SPACE.
	03	CARD-NAME		PIC X(10)	    VALUE  SPACE.
	03 	CARD-LIMIT			PIC S9(5)	    VALUE  ZERO.
	03 	CARD-DUE			PIC S9(05)V9(02)       VALUE    ZERO.

01 	WS-VALID-INPUT			PIC X(01)		VALUE SPACE.
01	WS-INREC-COUNT		PIC 9(04)		VALUE ZERO.
01 WS-OUT-REC-COUNT 	PIC 9(04)		VALUE ZERO.
01 	WS-END-INPUT			PIC X(01)		VALUE SPACE.
01 WS-DUP-CHECK			PIC X(05) 		VALUE SPACE.

PROCEDURE DIVISION.
000-MAIN-PARA.
		DISPLAY ‘CARD PROGRAM STARTED’
					
		PERFORM 1000-INITIAL-PARA THRU 1000-EXIT.
		PERFORM 2000-INPUT-PARA THRU 2000-EXIT.
		PERFORM 3000-PROCESS-PARA THRU 3000-EXIT.
								UNTIL WS-END-INPUT = ’Y’.

		DISPLAY   ‘TOTAL NUMBER OF INPUT’           WS-INREC-COUNT.
		DISPLAY   ‘TOTAL NUMBER OF OUTPUT’       WS-OUTREC-COUNT.
		STOP RUN.

1000-INITIAL-PARA 
		MOVE SPACE TO CARD-NUM OF CARD-DETAILS-IN.
		MOVE SPACE TO CARD-NAME OF CARD-DETAILS-IN.
		MOVE ZERO TO CARD-LIMIT OF CARD-DETAILS-IN.
		MOVE ZERO TO CARD-DUE OF CARD-DETAILS-IN.
		MOVE ’N’ TO WS-END-INPUT.
		MOVE SPACE TO WS-DUP-CHECK.
		MOVE ZERO TO WS-INREC-COUNT.
		MOVE ZERO TO WS-OUT-REC-COUNT.
				
	      
1000-EXIT. EXIT.

2000-INPUT-PARA.
		ACCEPT CARD-NUM OF CARD-DETAILS-IN.
		ACCEPT CARD-NAME OF CARD-DETAILS-IN.
		ACCEPT CARD-LIMIT OF CARD-DETAILS-IN.
		ACCEPT CARD-DUE OF CARD-DETAILS-IN.
			ADD +1 TO WS-INREC-COUNT.
2000-EXIT.

3000-PROCESS-PARA.

		MOVE ‘Y’ TO WS-VALID-INPUT.
		PERFORM 3100-AUDIT-PARA 	THRU      3100-EXIT.

		
		IF WS-VALID-INPUT = ‘Y’
			PERFORM 3110-DUP-CHECK-PARA    THRU   3110-EXIT
		END-IF

		IF WS-VALID-INPUT = ‘Y’
			PERFORM 3200-MOVE-PARA	 THRU     3200-EXIT
		ELSE 
				DISPLAY ‘DUPLICATE OR INVALID RECORD FOUND’
		END-IF
		
		ACCEPT WS-END-INPUT.

		IF WS-END-INPUT = ’N’
		PERFORM 2000-INPUT-PARA 	THRU      2000-EXIT
		END-IF.

3000-EXIT. EXIT.

3100-AUDIT-PARA.
				IF CARD-NUM OF CARD-DETAILS-IN = SPACE
					MOVE ’N’ TO WS-VALID-INPUT					GOTO 3100-EXIT	
				END-IF.
			
				IF CARD-NAME OF CARD-DETAILS-IN = SPACE
					MOVE ’N’ TO WS-VALID-INPUT					GOTO 3100-EXIT
				END-IF.
				
				IF CARD-LIMIT OF CARD-DETAILS-IN < 0
					MOVE ’N’ TO WS-VALID-INPUT					GOTO 3100-EXIT
				END-IF.
				
				IF CARD-DUE OF CARD-DETAILS-IN < 0
					MOVE ’N’ TO WS-VALID-INPUT
					GOTO 3100-EXIT
				END-IF.

3100-EXIT. EXIT.

3110-DUP-CHECK-PARA.

				IF CARD-NUM OF CARD-DETAILS-IN = WS-DUP-CHECK
					MOVE ’N’ TO WS-VALID-INPUT
					DISPLAY ‘DUPLICATE RECORD FOUND’
				END-IF

3200-MOVE-PARA.
		
		ADD +1 TO WS-OUT-REC-COUNT.
		MOVE CARD-NUM OF CARD-DETAILS-IN TO WS-DUP-CHECK.

		MOVE CARD-NUM OF CARD-DETAILS-IN    TO
				CARD-NUM OF CARD-DETAILS-OUT.
	
		MOVE CARD-NAME OF CARD-DETAILS-IN    TO
				CARD-NAME OF CARD-DETAILS-OUT.

		MOVE CARD-LIMIT OF CARD-DETAILS-IN    TO
				CARD-LIMIT OF CARD-DETAILS-OUT.

		MOVE CARD-DUE OF CARD-DETAILS-IN    TO
				CARD-DUE OF CARD-DETAILS-OUT.
		
		DISPLAY ‘VALID CARD DETAILS ARE’  CARD-DETAILS-OUT.

3200-EXIT. EXIT.
